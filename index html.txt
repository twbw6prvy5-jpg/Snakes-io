let AC = null;
function initAudio() {
  if (!AC) {
    try { AC = new (window.AudioContext || window.webkitAudioContext)(); } catch(e) { return; }
  }
  if (AC.state === 'suspended') AC.resume();
}
function beep(freq, dur, type, vol) {
  if (!AC) return;
  const o = AC.createOscillator(), g = AC.createGain();
  o.type = type || 'square';
  o.frequency.value = freq;
  g.gain.value = vol || 0.06;
  o.connect(g); g.connect(AC.destination);
  const t = AC.currentTime;
  g.gain.setValueAtTime(vol || 0.06, t);
  g.gain.exponentialRampToValueAtTime(0.001, t + dur);
  o.start(t); o.stop(t + dur);
}

class Food {
  constructor(x, y, color, r) {
    if (x === undefined) {
      const a = Math.random() * Math.PI * 2;
      const d = Math.sqrt(Math.random()) * (WORLD_R - 30);
      this.x = Math.cos(a) * d;
      this.y = Math.sin(a) * d;
    } else { this.x = x; this.y = y; }
    this.r = r || (2.5 + Math.random() * 2.5);
    this.color = color || COLORS[(Math.random() * COLORS.length) | 0];
    this.phase = Math.random() * 6.28;
  }
  draw() {
    this.phase += 0.06;
    const p = Math.sin(this.phase) * 1.4;
    ctx.beginPath();
    ctx.arc(this.x, this.y, this.r + p, 0, 6.283);
    ctx.fillStyle = this.color;
    ctx.shadowBlur = 12; ctx.shadowColor = this.color;
    ctx.fill();
    ctx.shadowBlur = 0;
  }
}

// ============ ЗМЕЙКА ============
class Snake {
  constructor(name, isBot) {
    this.name = name;
    this.isBot = isBot;
    this.color = isBot ? COLORS[(Math.random() * COLORS.length) | 0] : '#00ffcc';

    const a = Math.random() * Math.PI * 2;
    const d = Math.sqrt(Math.random()) * (WORLD_R * 0.75);
    this.x = Math.cos(a) * d;
    this.y = Math.sin(a) * d;
    this.angle = Math.random() * Math.PI * 2;
    this.radius = BASE_R;
    this.length = 10;
    this.path = [];
    this.isDead = false;
    this.boosting = false;

    // AI-поля
    this.target = { x: this.x, y: this.y };
    this.retargetIn = 0;
    this.hist = [];       // история позиций для детектора кружения
    this.stuckTimer = 0;  // счётчик «не двигаюсь»
    this.lastHistPos = { x: this.x, y: this.y };
  }

  update() {
    if (this.isDead) return;

    if (this.isBot) this.updateAI();
    else this.updatePlayer();

    const sp = (this.boosting && this.length > 15) ? BOOST_SPEED : SPEED;
    this.x += Math.cos(this.angle) * sp;
    this.y += Math.sin(this.angle) * sp;

    // граница мира — плавно отталкиваем внутрь
    const dc = Math.hypot(this.x, this.y);
    if (dc > WORLD_R - this.radius) {
      const a = Math.atan2(this.y, this.x);
      this.x = Math.cos(a) * (WORLD_R - this.radius);
      this.y = Math.sin(a) * (WORLD_R - this.radius);
      // развернуть к центру
      const toCenter = Math.atan2(-this.y, -this.x);
      let diff = toCenter - this.angle;
      while (diff > Math.PI) diff -= 6.283;
      while (diff < -Math.PI) diff += 6.283;
      this.angle += diff * 0.3;
    }

    // тропа тела
    if (!this.path.length ||
        Math.hypot(this.x - this.path[0].x, this.y - this.path[0].y) > SEG_GAP) {
      this.path.unshift({ x: this.x, y: this.y });
      while (this.path.length > this.length) this.path.pop();
    }

    // еда
    for (let i = foods.length - 1; i >= 0; i--) {
      const f = foods[i];
      const d = Math.hypot(this.x - f.x, this.y - f.y);
      if (d < this.radius + f.r + 4) {
        foods.splice(i, 1);
        this.length += 1;
        if (this.radius < 28) this.radius += 0.08;
        foods.push(new Food());
        if (!this.isBot) beep(500 + Math.random() * 200, 0.06, 'square', 0.05);
      }
    }
  }

  updatePlayer() {
    if (joyActive && (joyVec.x || joyVec.y)) {
      const targetAngle = Math.atan2(joyVec.y, joyVec.x);
      let diff = targetAngle - this.angle;
      while (diff > Math.PI) diff -= 6.283;
      while (diff < -Math.PI) diff += 6.283;
      this.angle += diff * 0.14;
    }
    this.boosting = boosting && this.length > 15;
  }

  updateAI() {
    this.retargetIn--;

    // ---- ДЕТЕКТОР КРУЖЕНИЯ ----
    // Каждые ~15 кадров запоминаем позицию. Если за 4 снимка сдвинулись мало — считаем что кружимся.
    this.stuckTimer++;
    if (this.stuckTimer >= 15) {
      this.stuckTimer = 0;
      this.hist.push({ x: this.x, y: this.y });
      if (this.hist.length > 4) this.hist.shift();
      if (this.hist.length === 4) {
        const a = this.hist[0], b = this.hist[3];
        const moved = Math.hypot(b.x - a.x, b.y - a.y);
        if (moved < 40) {
          // застрял в петле — срочно меняем цель на случайную точку подальше
          this.pickEscapeTarget();
          this.hist = [];
          this.retargetIn = 40;
        }
      }
    }

    if (this.retargetIn <= 0) {
      this.retargetIn = 25 + Math.random() * 20;
      this.chooseTarget();
    }

    // поворот к цели
    const dx = this.target.x - this.x;
    const dy = this.target.y - this.y;
    const targetAngle = Math.atan2(dy, dx);
    let diff = targetAngle - this.angle;
    while (diff > Math.PI) diff -= 6.283;
    while (diff < -Math.PI) diff += 6.283;

    // быстрый поворот = маленький радиус разворота = не кружит
    this.angle += diff * 0.11;

    this.boosting = this.length > 25 && Math.hypot(dx, dy) > 250 && Math.abs(diff) < 0.4;
  }

  chooseTarget() {
    // 1. агрессия к игроку если крупный и близко
    if (this.length > 30 && player && !player.isDead) {
      const dp = Math.hypot(this.x - player.x, this.y - player.y);
      if (dp < 350) {
        this.target = { x: player.x, y: player.y };
        return;
      }
    }

    // 2. ищем лучшую еду: близко + не сзади
    let best = null, bestScore = Infinity;
    for (let f of foods) {
      const dx = f.x - this.x, dy = f.y - this.y;
      const d = Math.hypot(dx, dy);
      if (d > 500 || d < 1) continue;

      let ad = Math.atan2(dy, dx) - this.angle;
      while (ad > Math.PI) ad -= 6.283;
      while (ad < -Math.PI) ad += 6.283;

      // штраф за угол: еда сзади невыгодна (иначе будет кружить)
      const score = d + Math.abs(ad) * 70;
      if (score < bestScore) { bestScore = score; best = f; }
    }

    if (best) {
      // целимся ВПЕРЁД еды, чтобы проскочить её телом, а не кружить вокруг
      const dx = best.x - this.x, dy = best.y - this.y;
      const d = Math.hypot(dx, dy) || 1;
      const lead = Math.min(d * 0.5, 40);
      this.target = { x: best.x + (dx / d) * lead, y: best.y + (dy / d) * lead };
    } else {
      this.pickEscapeTarget();
    }
  }

  pickEscapeTarget() {
    // случайная точка не слишком далеко от нас и не у самой стены
    for (let tries = 0; tries < 8; tries++) {
      const a = Math.random() * 6.283;
      const dist = 200 + Math.random() * 250;
      const tx = this.x + Math.cos(a) * dist;
      const ty = this.y + Math.sin(a) * dist;
      if (Math.hypot(tx, ty) < WORLD_R - 100) {
        this.target = { x: tx, y: ty };
        return;
      }
    }
    // фолбэк — к центру
    this.target = { x: 0, y: 0 };
  }

  draw() {
    if (this.isDead) return;

    // тело
    for (let i = this.path.length - 1; i >= 0; i--) {
      const p = this.path[i];
      const t = i / Math.max(1, this.length);
      const r = this.radius * (1 - t * 0.45);
      if (r < 0.5) continue;
      ctx.beginPath();
      ctx.arc(p.x, p.y, r, 0, 6.283);
      ctx.fillStyle = this.color;
      ctx.shadowBlur = this.boosting ? 22 : 8;
      ctx.shadowColor = this.color;
      ctx.fill();
    }
    ctx.shadowBlur = 0;

    // голова
    ctx.beginPath();
    ctx.arc(this.x, this.y, this.radius, 0, 6.283);
    ctx.fillStyle = '#fff';
    ctx.fill();

    // глаза
    const ex = Math.cos(this.angle) * this.radius * 0.45;
    const ey = Math.sin(this.angle) * this.radius * 0.45;
    const px = -Math.sin(this.angle) * this.radius * 0.35;
    const py = Math.cos(this.angle) * this.radius * 0.35;
    ctx.fillStyle = '#000';
    ctx.beginPath();
    ctx.arc(this.x + ex + px, this.y + ey + py, this.radius * 0.28, 0, 6.283);
    ctx.fill();
    ctx.beginPath();
    ctx.arc(this.x + ex - px, this.y + ey - py, this.radius * 0.28, 0, 6.283);
    ctx.fill();

    // имя только у игрока
    if (!this.isBot) {
      ctx.fillStyle = '#00ffcc';
      ctx.font = 'bold 13px sans-serif';
      ctx.textAlign = 'center';
      ctx.fillText(this.name, this.x, this.y - this.radius - 8);
    }
  }
}

// ============ СОСТОЯНИЕ ИГРЫ ============
let foods = [];
let bots = [];
let player = null;
let playing = false;
const camera = { x: 0, y: 0 };

function newFoodAt(x, y, color) {
  foods.push(new Food(x, y, color, 3 + Math.random() * 2));
}

function initGame() {
  foods = [];
  bots = [];
  player = null;

  for (let i = 0; i < FOOD_N; i++) foods.push(new Food());

  const nick = (document.getElementById('nick').value || 'Игрок').slice(0, 10);
  player = new Snake(nick, false);

  const names = ['Змей','Кобра','Гадюка','Удав','Питон','Анаконда','Уж','Вивер','Жук'];
  for (let i = 0; i < BOT_N; i++) {
    const n = names[(Math.random() * names.length) | 0] + ' ' + (i + 1);
    bots.push(new Snake(n, true));
  }

  camera.x = player.x;
  camera.y = player.y;
  playing = true;
  setBoost(false);
}

// ============ СТОЛКНОВЕНИЯ ============
function checkCollisions() {
  if (!player || player.isDead) return;

  for (let bi = bots.length - 1; bi >= 0; bi--) {
    const bot = bots[bi];
    if (bot.isDead) continue;

    // игрок врезался в бота?
    let hit = false;
    const step = 2;
    for (let i = 0; i < bot.path.length; i += step) {
      const p = bot.path[i];
      if (Math.hypot(player.x - p.x, player.y - p.y) < player.radius + bot.radius * 0.75) {
        hit = true; break;
      }
    }
    if (hit) { killPlayer(); return; }

    // бот врезался в игрока?
    hit = false;
    for (let i = 0; i < player.path.length; i += step) {
      const p = player.path[i];
      if (Math.hypot(bot.x - p.x, bot.y - p.y) < bot.radius + player.radius * 0.75) {
        hit = true; break;
      }
    }
    if (hit) {
      bot.isDead = true;
      beep(800, 0.15, 'triangle', 0.07);
      for (let j = 0; j < Math.floor(bot.length / 2); j++) {
        newFoodAt(bot.x + (Math.random() - 0.5) * 50,
                  bot.y + (Math.random() - 0.5) * 50, bot.color);
      }
      // респавн через паузу
      setTimeout(() => {
        if (!playing) return;
        const names = ['Змей','Кобра','Гадюка','Удав','Питон'];
        const n = names[(Math.random() * names.length) | 0] + ' ' + ((Math.random() * 100) | 0);
        bots.push(new Snake(n, true));
      }, 2500);
      bots.splice(bi, 1);
    }
  }
}

function killPlayer() {
  player.isDead = true;
  playing = false;
  beep(180, 0.6, 'sawtooth', 0.15);
  document.getElementById('finalLen').textContent = 'Длина: ' + Math.floor(player.length);
  document.getElementById('over').style.display = 'flex';
  for (let j = 0; j < Math.floor(player.length / 2); j++) {
    newFoodAt(player.x + (Math.random() - 0.5) * 50,
              player.y + (Math.random() - 0.5) * 50, player.color);
  }
  endJoy();
  setBoost(false);
}

// ============ ЛИДЕРБОРД ============
function updateBoard() {
  const all = [player, ...bots].filter(s => s && !s.isDead);
  all.sort((a, b) => b.length - a.length);
  let html = '';
  for (let i = 0; i < Math.min(5, all.length); i++) {
    const s = all[i];
    const cls = s === player ? ' class="me"' : '';
    html += `<div${cls}>${i + 1}. ${s.name}: ${Math.floor(s.length)}</div>`;
  }
  document.getElementById('board').innerHTML = html;
  if (player && !player.isDead) {
    document.getElementById('score').textContent = 'Длина: ' + Math.floor(player.length);
  }
}

// ============ РЕНДЕР ============
function render() {
  ctx.fillStyle = '#0a0a0f';
  ctx.fillRect(0, 0, W, H);

  const camX = player ? camera.x : 0;
  const camY = player ? camera.y : 0;

  ctx.save();
  ctx.translate(W / 2 - camX, H / 2 - camY);

  // граница
  ctx.beginPath();
  ctx.arc(0, 0, WORLD_R, 0, 6.283);
  ctx.strokeStyle = 'rgba(255,51,102,.6)';
  ctx.lineWidth = 8;
  ctx.shadowBlur = 40; ctx.shadowColor = '#ff3366';
  ctx.stroke();
  ctx.shadowBlur = 0;

  // сетка
  const gs = 120;
  const x0 = Math.floor((camX - W / 2) / gs) * gs;
  const y0 = Math.floor((camY - H / 2) / gs) * gs;
  ctx.strokeStyle = 'rgba(255,255,255,.04)';
  ctx.lineWidth = 1;
  for (let x = x0; x < camX + W / 2 + gs; x += gs) {
    ctx.beginPath(); ctx.moveTo(x, camY - H / 2); ctx.lineTo(x, camY + H / 2); ctx.stroke();
  }
  for (let y = y0; y < camY + H / 2 + gs; y += gs) {
    ctx.beginPath(); ctx.moveTo(camX - W / 2, y); ctx.lineTo(camX + W / 2, y); ctx.stroke();
  }

  // еда (только в видимой области)
  const l = camX - W / 2 - 40, r = camX + W / 2 + 40;
  const t = camY - H / 2 - 40, b = camY + H / 2 + 40;
  for (let f of foods) {
    if (f.x > l && f.x < r && f.y > t && f.y < b) f.draw();
  }

  // боты (мелкие сначала — крупные сверху)
  bots.sort((a, b) => a.length - b.length);
  for (let bot of bots) if (!bot.isDead) bot.draw();

  // игрок
  if (player && !player.isDead) player.draw();

  ctx.restore();
}

function loop() {
  requestAnimationFrame(loop);

  if (playing && player && !player.isDead) {
    player.update();

    // плавная камера
    camera.x += (player.x - camera.x) * 0.12;
    camera.y += (player.y - camera.y) * 0.12;

    for (let bot of bots) if (!bot.isDead) bot.update();

    checkCollisions();
    updateBoard();
  }

  render();
}
